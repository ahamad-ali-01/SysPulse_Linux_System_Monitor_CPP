cmake_minimum_required(VERSION 3.16)
project(SysPulse VERSION 1.0.0 LANGUAGES CXX)
option(SYSPULSE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(syspulse_core src/monitor.cpp src/exporter.cpp src/cli.cpp)
target_include_directories(syspulse_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(syspulse_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(SYSPULSE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(syspulse_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(syspulse_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(syspulse src/main.cpp)
target_link_libraries(syspulse PRIVATE syspulse_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(syspulse_tests tests/test_main.cpp)
  target_link_libraries(syspulse_tests PRIVATE syspulse_core)
  add_test(NAME syspulse_unit_tests COMMAND syspulse_tests)
  add_test(NAME syspulse_help COMMAND syspulse --help)
  add_test(NAME syspulse_invalid_option COMMAND syspulse processes --limit 0)
  set_tests_properties(syspulse_invalid_option PROPERTIES WILL_FAIL TRUE)
  add_test(NAME syspulse_memory COMMAND syspulse memory)
  add_test(NAME syspulse_processes COMMAND syspulse processes --sort pid --limit 5)
endif()
install(TARGETS syspulse RUNTIME DESTINATION bin)
